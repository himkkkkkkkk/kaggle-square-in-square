# Squares in Squares：判别器与打分器

本交付实现竞赛的合法性判别、边长测量、单题得分、缺失或非法布局的网格罚分，以及总分计算。几何核心保留十进制字符串，使用浮点 SAT 初筛和 `mpmath` 区间算术认证，不用浮点容差接受面积重叠。

## 数学契约

对于每个 \(n\)，从所有旋转后的顶点计算

\[
L_n=\max(\max X-\min X,\max Y-\min Y),
\qquad s_n=\frac{L_n}{\sqrt n}-1.
\]

非法或缺失布局使用

\[
s_n^{\mathrm{grid}}=\frac{\lceil\sqrt n\rceil}{\sqrt n}-1.
\]

## 运行

```bash
python3 -m pip install -r requirements.txt
python3 -m unittest discover -s tests -v
python3 -m squares_metric submission.csv --details submission_details.csv
```

程序输出总分、合法题目数，并写出逐题的 `n, valid, L, score, reason`。

## 直接调用

```python
from squares_metric import Square, evaluate_squares, score_submission_file

result = evaluate_squares(
```

```python
from __future__ import annotations

import contextlib
import math
from collections.abc import Sequence
from dataclasses import dataclass
from itertools import takewhile

import mpmath

MEASUREMENT_DPS = 60
CERTIFICATION_DPS = (50, 100, 200, 500)
UNIT_SQUARE_HALF_SIDE = 0.5
BROAD_PHASE_PADDING = 1e-7
FLOAT_CONTACT_TOLERANCE = 1e-8

Interval = mpmath.ctx_iv.ivmpf
Point = tuple[float, float]
IntervalPoint = tuple[Interval, Interval]
FloatCorners = tuple[Point, Point, Point, Point]
IntervalCorners = tuple[IntervalPoint, IntervalPoint, IntervalPoint, IntervalPoint]


@dataclass(frozen=True)
class Square:
    x: str
    y: str
    angle: str


@dataclass(frozen=True)
class FloatSquare:
    corners: FloatCorners
    angle: float


@contextlib.contextmanager
def working_precision(dps: int):
    previous_mp_dps = mpmath.mp.dps
    previous_iv_dps = mpmath.iv.dps
    try:
        mpmath.mp.dps = dps
        mpmath.iv.dps = dps
        yield
    finally:
        mpmath.mp.dps = previous_mp_dps
        mpmath.iv.dps = previous_iv_dps


def float_corners(x: float, y: float, angle: float) -> FloatCorners:
    cosine = math.cos(angle)
    sine = math.sin(angle)
    first_dx = UNIT_SQUARE_HALF_SIDE * (cosine - sine)
    first_dy = UNIT_SQUARE_HALF_SIDE * (sine + cosine)
    second_dx = UNIT_SQUARE_HALF_SIDE * (-cosine - sine)
    second_dy = UNIT_SQUARE_HALF_SIDE * (-sine + cosine)
    return (
        (x + first_dx, y + first_dy),
        (x + second_dx, y + second_dy),
        (x - first_dx, y - first_dy),
        (x - second_dx, y - second_dy),
    )


def prepare_float_squares(squares: Sequence[Square]) -> tuple[FloatSquare, ...]:
    with working_precision(MEASUREMENT_DPS):
        origin_x = mpmath.mpf(squares[0].x)
        origin_y = mpmath.mpf(squares[0].y)
        full_turn = 2 * mpmath.pi
        poses = tuple(
            (
                float(mpmath.mpf(square.x) - origin_x),
                float(mpmath.mpf(square.y) - origin_y),
                float(mpmath.mpf(square.angle) % full_turn),
            )
            for square in squares
        )
    return tuple(FloatSquare(float_corners(x, y, angle), angle) for x, y, angle in poses)


def measure_side_length(squares: Sequence[Square], dps: int = MEASUREMENT_DPS) -> mpmath.mpf:
    with working_precision(dps):
        origin_x = mpmath.mpf(squares[0].x)
        origin_y = mpmath.mpf(squares[0].y)
        corners = tuple(
            corner
            for square in squares
            for corner in mp_corners(
                mpmath.mpf(square.x) - origin_x,
                mpmath.mpf(square.y) - origin_y,
                mpmath.mpf(square.angle),
            )
        )
        x_coordinates = tuple(x for x, _ in corners)
        y_coordinates = tuple(y for _, y in corners)
        return max(
            max(x_coordinates) - min(x_coordinates),
            max(y_coordinates) - min(y_coordinates),
        )


def mp_corners(x: mpmath.mpf, y: mpmath.mpf, angle: mpmath.mpf):
    cosine = mpmath.cos(angle)
    sine = mpmath.sin(angle)
    half_side = mpmath.mpf(1) / 2
    first_dx = half_side * (cosine - sine)
    first_dy = half_side * (sine + cosine)
    second_dx = half_side * (-cosine - sine)
    second_dy = half_side * (-sine + cosine)
    return (
        (x + first_dx, y + first_dy),
        (x + second_dx, y + second_dy),
        (x - first_dx, y - first_dy),
        (x - second_dx, y - second_dy),
    )


def interval_corners(square: Square) -> IntervalCorners:
    x = mpmath.iv.mpf(square.x)
    y = mpmath.iv.mpf(square.y)
    angle = mpmath.iv.mpf(square.angle)
    cosine = mpmath.iv.cos(angle)
    sine = mpmath.iv.sin(angle)
    half_side = mpmath.iv.mpf("0.5")
    first_dx = half_side * (cosine - sine)
    first_dy = half_side * (sine + cosine)
    second_dx = half_side * (-cosine - sine)
    second_dy = half_side * (-sine + cosine)
    return (
        (x + first_dx, y + first_dy),
        (x + second_dx, y + second_dy),
        (x - first_dx, y - first_dy),
        (x - second_dx, y - second_dy),
    )


def candidate_pairs(squares: Sequence[FloatSquare]) -> tuple[tuple[int, int], ...]:
    bounds = tuple(enumerate(tuple(padded_bounds(square.corners) for square in squares)))
    ordered_bounds = tuple(sorted(bounds, key=lambda item: item[1][0]))
    return tuple(
        (min(first_index, second_index), max(first_index, second_index))
        for first_position, (first_index, first_bounds) in enumerate(ordered_bounds)
        for second_index, second_bounds in takewhile(
            lambda item: item[1][0] <= first_bounds[2],
            ordered_bounds[first_position + 1 :],
        )
        if bounds_overlap(first_bounds, second_bounds)
    )


def padded_bounds(corners: FloatCorners) -> tuple[float, float, float, float]:
    x_coordinates = tuple(x for x, _ in corners)
    y_coordinates = tuple(y for _, y in corners)
    return (
        min(x_coordinates) - BROAD_PHASE_PADDING,
        min(y_coordinates) - BROAD_PHASE_PADDING,
        max(x_coordinates) + BROAD_PHASE_PADDING,
        max(y_coordinates) + BROAD_PHASE_PADDING,
    )


def bounds_overlap(first: tuple[float, float, float, float], second: tuple[float, float, float, float]) -> bool:
    return not (
        first[2] < second[0]
        or second[2] < first[0]
        or first[3] < second[1]
        or second[3] < first[1]
    )


def float_pair_status(first: FloatSquare, second: FloatSquare) -> str:
    axes = separation_axes(first.angle, second.angle)
    projections = tuple(
        projection_overlap(first.corners, second.corners, axis)
        for axis in axes
    )
    if min(projections) < -FLOAT_CONTACT_TOLERANCE:
        return "separated"
    if min(projections) > FLOAT_CONTACT_TOLERANCE:
        return "overlapping"
    return "undecided"


def separation_axes(first_angle: float, second_angle: float) -> tuple[Point, ...]:
    first_cosine, first_sine = math.cos(first_angle), math.sin(first_angle)
    second_cosine, second_sine = math.cos(second_angle), math.sin(second_angle)
    return (
        (first_cosine, first_sine),
        (-first_sine, first_cosine),
        (second_cosine, second_sine),
        (-second_sine, second_cosine),
    )


def projection_overlap(first: FloatCorners, second: FloatCorners, axis: Point) -> float:
    axis_x, axis_y = axis
    first_projections = tuple(axis_x * x + axis_y * y for x, y in first)
    second_projections = tuple(axis_x * x + axis_y * y for x, y in second)
    return min(max(first_projections), max(second_projections)) - max(
        min(first_projections), min(second_projections)
    )


def certified_pair_separated(first: Square, second: Square) -> bool | None:
    first_corners = interval_corners(first)
    second_corners = interval_corners(second)
    first_angle = mpmath.iv.mpf(first.angle)
    second_angle = mpmath.iv.mpf(second.angle)
    first_cosine, first_sine = mpmath.iv.cos(first_angle), mpmath.iv.sin(first_angle)
    second_cosine, second_sine = mpmath.iv.cos(second_angle), mpmath.iv.sin(second_angle)
    axes = (
        (first_cosine, first_sine),
        (-first_sine, first_cosine),
        (second_cosine, second_sine),
        (-second_sine, second_cosine),
    )
    undecided_axis_found = False
    for axis_x, axis_y in axes:
        first_projections = tuple(axis_x * x + axis_y * y for x, y in first_corners)
        second_projections = tuple(axis_x * x + axis_y * y for x, y in second_corners)
        first_minimum, first_maximum = interval_span(first_projections)
        second_minimum, second_maximum = interval_span(second_projections)
        overlap = interval_min((first_maximum, second_maximum)) - interval_max(
            (first_minimum, second_minimum)
        )
        comparison = interval_less_or_equal(overlap, mpmath.iv.mpf("0"))
        if comparison is True:
            return True
        if comparison is None:
            undecided_axis_found = True
    return None if undecided_axis_found else False


def interval_span(values: Sequence[Interval]) -> tuple[Interval, Interval]:
    return interval_min(values), interval_max(values)


def interval_min(values: Sequence[Interval]) -> Interval:
    return mpmath.iv.mpf((min(value.a for value in values), min(value.b for value in values)))


def interval_max(values: Sequence[Interval]) -> Interval:
    return mpmath.iv.mpf((max(value.a for value in values), max(value.b for value in values)))


def interval_less_or_equal(first: Interval, second: Interval) -> bool | None:
    try:
        return first <= second
    except ValueError:
        return None


def certify_pair_separation(first: Square, second: Square) -> bool | None:
    for dps in CERTIFICATION_DPS:
        with working_precision(dps):
            separated = certified_pair_separated(first, second)
        if separated is not None:
            return separated
    return None
```

### `squares_metric/validation.py`

```python
from __future__ import annotations

import csv
import io
import math
import re
from collections import defaultdict
from collections.abc import Sequence

from .geometry import (
    Square,
    candidate_pairs,
    certify_pair_separation,
    float_pair_status,
    prepare_float_squares,
)

MAXIMUM_PUZZLE_SIZE = 1000
MAXIMUM_COORDINATE_MAGNITUDE = 1e4
MAXIMUM_ANGLE_MAGNITUDE = 100.0
MAXIMUM_DECIMAL_LITERAL_LENGTH = 100
REQUIRED_SUBMISSION_COLUMNS = frozenset(("n", "square_id", "x", "y", "angle"))
DECIMAL_LITERAL = re.compile(r"^[+-]?(\d+\.?\d*|\.\d+)([eE][+-]?\d+)?$")


def parse_submission(text: str) -> dict[int, tuple[dict[str, str], ...]]:
    reader = csv.DictReader(io.StringIO(text))
    missing_columns = REQUIRED_SUBMISSION_COLUMNS.difference(reader.fieldnames or ())
    if missing_columns:
        raise ValueError(f"missing columns: {sorted(missing_columns)}")
    rows_by_puzzle = defaultdict(list)
    for row_number, row in enumerate(reader, start=2):
        puzzle_size = int(row["n"])
        if not 1 <= puzzle_size <= MAXIMUM_PUZZLE_SIZE:
            raise ValueError(f"row {row_number}: n={puzzle_size} is outside 1..{MAXIMUM_PUZZLE_SIZE}")
        rows_by_puzzle[puzzle_size].append(row)
    return {n: tuple(rows) for n, rows in rows_by_puzzle.items()}


def puzzle_structure_error(n: int, rows: Sequence[dict[str, str]]) -> str | None:
    if len(rows) != n:
        return f"expected {n} rows, received {len(rows)}"
    square_ids = {int(row["square_id"]) for row in rows}
    expected_ids = set(range(n))
    if square_ids != expected_ids:
        return f"square_id values must be 0..{n - 1}"
    return None


def squares_from_rows(n: int, rows: Sequence[dict[str, str]]) -> tuple[Square, ...]:
    rows_by_id = {int(row["square_id"]): row for row in rows}
    return tuple(
        Square(
            rows_by_id[square_id]["x"].strip(),
            rows_by_id[square_id]["y"].strip(),
            rows_by_id[square_id]["angle"].strip(),
        )
        for square_id in range(n)
    )


def validate_decimal_literals(square: Square, square_id: int) -> None:
    for field_name, literal in (("x", square.x), ("y", square.y), ("angle", square.angle)):
        if len(literal) > MAXIMUM_DECIMAL_LITERAL_LENGTH or not DECIMAL_LITERAL.fullmatch(literal):
            raise ValueError(f"square {square_id}: {field_name} is not a decimal literal")
        value = float(literal)
        if not math.isfinite(value):
            raise ValueError(f"square {square_id}: {field_name} is outside float range")
    if max(abs(float(square.x)), abs(float(square.y))) > MAXIMUM_COORDINATE_MAGNITUDE:
        raise ValueError(f"square {square_id}: coordinate magnitude exceeds {MAXIMUM_COORDINATE_MAGNITUDE:g}")
    if abs(float(square.angle)) > MAXIMUM_ANGLE_MAGNITUDE:
        raise ValueError(f"square {square_id}: angle magnitude exceeds {MAXIMUM_ANGLE_MAGNITUDE:g}")


def verify_no_overlaps(squares: Sequence[Square]) -> tuple[bool, str]:
    float_squares = prepare_float_squares(squares)
    undecided_pairs = []
    for first_id, second_id in candidate_pairs(float_squares):
        status = float_pair_status(float_squares[first_id], float_squares[second_id])
        if status == "overlapping":
            return False, f"squares {first_id} and {second_id} overlap"
        if status == "undecided":
            undecided_pairs.append((first_id, second_id))
    for first_id, second_id in undecided_pairs:
        separated = certify_pair_separation(squares[first_id], squares[second_id])
        if separated is False:
            return False, f"squares {first_id} and {second_id} overlap"
        if separated is None:
            return False, f"squares {first_id} and {second_id} are undecidable at maximum precision"
    return True, "feasible"
```

### `squares_metric/scoring.py`

```python
from __future__ import annotations

import math
from collections.abc import Iterable, Sequence
from dataclasses import dataclass
from pathlib import Path

import mpmath

from .geometry import Square, measure_side_length
from .validation import (
    parse_submission,
    puzzle_structure_error,
    squares_from_rows,
    validate_decimal_literals,
    verify_no_overlaps,
)

AREA_LOWER_BOUND_TOLERANCE = 1e-7
DEFAULT_PUZZLE_SIZES = tuple(range(1, 1001))


@dataclass(frozen=True)
class PuzzleResult:
    n: int
    valid: bool
    score: float
    side_length: mpmath.mpf | None
    reason: str


@dataclass(frozen=True)
class ScoreReport:
    total_score: float
    puzzles: tuple[PuzzleResult, ...]


def score_from_side_length(n: int, side_length: float) -> float:
    return side_length / math.sqrt(n) - 1.0


def grid_fallback_score(n: int) -> float:
    return math.ceil(math.sqrt(n)) / math.sqrt(n) - 1.0


def invalid_puzzle_result(n: int, reason: str) -> PuzzleResult:
    return PuzzleResult(n, False, grid_fallback_score(n), None, reason)


def evaluate_squares(n: int, squares: Sequence[Square]) -> PuzzleResult:
    if len(squares) != n:
        return invalid_puzzle_result(n, f"expected {n} squares, received {len(squares)}")
    try:
        for square_id, square in enumerate(squares):
            validate_decimal_literals(square, square_id)
    except ValueError as error:
        return invalid_puzzle_result(n, str(error))
    feasible, reason = verify_no_overlaps(squares)
    if not feasible:
        return invalid_puzzle_result(n, reason)
    side_length = measure_side_length(squares)
    if float(side_length) < math.sqrt(n) - AREA_LOWER_BOUND_TOLERANCE:
        return invalid_puzzle_result(n, "bounding side is below the area lower bound")
    puzzle_score = max(0.0, score_from_side_length(n, float(side_length)))
    return PuzzleResult(n, True, puzzle_score, side_length, "feasible")


def evaluate_puzzle(n: int, rows: Sequence[dict[str, str]]) -> PuzzleResult:
    try:
        structure_error = puzzle_structure_error(n, rows)
        squares = squares_from_rows(n, rows) if structure_error is None else ()
    except ValueError as error:
        return invalid_puzzle_result(n, str(error))
    if structure_error:
        return invalid_puzzle_result(n, structure_error)
    return evaluate_squares(n, squares)


def score_submission_text(text: str, puzzle_sizes: Iterable[int] = DEFAULT_PUZZLE_SIZES) -> ScoreReport:
    requested_sizes = frozenset(puzzle_sizes)
    submitted_rows = parse_submission(text)
    results = tuple(
        evaluate_puzzle(n, submitted_rows[n]) if n in submitted_rows else invalid_puzzle_result(n, "missing")
        for n in sorted(requested_sizes)
    )
    return ScoreReport(sum(result.score for result in results), results)


def score_submission_file(
    submission_path: str | Path,
    puzzle_sizes: Iterable[int] = DEFAULT_PUZZLE_SIZES,
) -> ScoreReport:
    return score_submission_text(Path(submission_path).read_text(), puzzle_sizes)
```

### `squares_metric/cli.py`

```python
from __future__ import annotations

import argparse
import csv
from pathlib import Path

import mpmath

from .scoring import score_submission_file

SIDE_LENGTH_DECIMAL_PLACES = 50


def main() -> None:
    parser = argparse.ArgumentParser(description="Validate and score a Squares in Squares submission.")
    parser.add_argument("submission", type=Path)
    parser.add_argument("--details", type=Path, default=Path("submission_details.csv"))
    arguments = parser.parse_args()
    report = score_submission_file(arguments.submission)
    write_details(arguments.details, report.puzzles)
    print(f"total_score={report.total_score:.12f}")
    print(f"valid_puzzles={sum(result.valid for result in report.puzzles)}/{len(report.puzzles)}")
    print(f"details={arguments.details}")


def write_details(path: Path, puzzles) -> None:
    with path.open("w", newline="") as output_file:
        writer = csv.writer(output_file)
        writer.writerow(("n", "valid", "L", "score", "reason"))
        for result in puzzles:
            side_length = (
                mpmath.nstr(result.side_length, SIDE_LENGTH_DECIMAL_PLACES + 2)
                if result.side_length is not None
                else ""
            )
            writer.writerow((result.n, result.valid, side_length, result.score, result.reason))


if __name__ == "__main__":
    main()
```

### `tests/test_metric.py`

```python
import math
import unittest

from squares_metric import (
    Square,
    evaluate_squares,
    grid_fallback_score,
    measure_side_length,
    score_submission_text,
    verify_no_overlaps,
)


def grid_rows(n):
    grid_width = math.ceil(math.sqrt(n))
    return [
        {
            "n": str(n),
            "square_id": str(square_id),
            "x": str(square_id % grid_width + 0.5),
            "y": str(square_id // grid_width + 0.5),
            "angle": "0",
        }
        for square_id in range(n)
    ]


def submission_text(puzzles):
    lines = ["n,square_id,x,y,angle"]
    for rows in puzzles:
        lines.extend(
            f"{row['n']},{row['square_id']},{row['x']},{row['y']},{row['angle']}"
            for row in rows
        )
    return "\n".join(lines)


class GeometryTests(unittest.TestCase):
    def test_unit_square_has_unit_side(self):
        self.assertEqual(measure_side_length((Square("0.5", "0.5", "0"),)), 1)

    def test_rotated_square_has_sqrt_two_side(self):
        square = Square("0", "0", "0.78539816339744830961566084581987572104929234984378")
        self.assertAlmostEqual(float(measure_side_length((square,))), math.sqrt(2), places=14)

    def test_translation_does_not_change_side(self):
        squares = (Square("0.5", "0.5", "0"), Square("1.5", "0.5", "0"))
        translated = (Square("1000.5", "-2000.5", "0"), Square("1001.5", "-2000.5", "0"))
        self.assertEqual(measure_side_length(squares), measure_side_length(translated))

    def test_edge_contact_is_legal(self):
        squares = (Square("0.5", "0.5", "0"), Square("1.5", "0.5", "0"))
        self.assertEqual(verify_no_overlaps(squares), (True, "feasible"))

    def test_area_overlap_is_illegal(self):
        squares = (Square("0.5", "0.5", "0"), Square("0.6", "0.5", "0"))
        self.assertFalse(verify_no_overlaps(squares)[0])

    def test_tiny_overlap_is_illegal(self):
        squares = (Square("0", "0", "0"), Square("0.99999999999999999999999999999999999999999999999999", "0", "0"))
        self.assertFalse(verify_no_overlaps(squares)[0])

    def test_separated_rotated_squares_are_legal(self):
        angle = "0.78539816339744830961566084581987572104929234984378"

class ScoringTests(unittest.TestCase):
    def test_perfect_square_grid_scores_zero(self):
        result = evaluate_squares(4, tuple(Square(row["x"], row["y"], row["angle"]) for row in grid_rows(4)))
        self.assertTrue(result.valid)
        self.assertEqual(result.score, 0)

    def test_two_square_grid_matches_formula(self):
        result = evaluate_squares(2, tuple(Square(row["x"], row["y"], row["angle"]) for row in grid_rows(2)))
        self.assertAlmostEqual(result.score, math.sqrt(2) - 1)

    def test_overlap_receives_grid_fallback(self):
        result = evaluate_squares(2, (Square("0", "0", "0"), Square("0", "0", "0")))
        self.assertFalse(result.valid)
        self.assertIsNone(result.side_length)
        self.assertEqual(result.score, grid_fallback_score(2))

        report = score_submission_text(submission_text((grid_rows(1),)), puzzle_sizes=(1, 2))
        self.assertEqual(report.puzzles[0].score, 0)
        self.assertEqual(report.puzzles[1].score, grid_fallback_score(2))
        self.assertEqual(report.total_score, grid_fallback_score(2))

        self.assertFalse(result.valid)
        self.assertEqual(result.score, grid_fallback_score(2))
```
